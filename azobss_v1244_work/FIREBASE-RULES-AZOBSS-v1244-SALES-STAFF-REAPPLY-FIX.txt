rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return request.auth != null
        && (request.auth.token.email == "zedan91@azobss.local" || request.auth.token.email == "zedan9107@gmail.com");
    }

    function isThisUser(username) {
      return request.auth != null
        && exists(/databases/$(database)/documents/users/$(username))
        && request.auth.uid == get(/databases/$(database)/documents/users/$(username)).data.uid;
    }


    // AZOBSS v1238: staff identity helper used by staff submission collections.
    function isStaffUser(username) {
      return request.auth != null
        && username is string
        && exists(/databases/$(database)/documents/users/$(username))
        && request.auth.uid == get(/databases/$(database)/documents/users/$(username)).data.uid
        && get(/databases/$(database)/documents/users/$(username)).data.role in [
          "staff", "Staff", "manager", "Manager",
          "semiAdmin", "semiadmin", "semi-admin", "semi_admin",
          "seller", "editor"
        ];
    }

    match /users/{username} {
      allow get: if true;
      allow list: if false;

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow update: if isAdmin()
        || isThisUser(username)
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
          && request.resource.data.uid == request.auth.uid
        );

      allow read, delete: if isAdmin() || isThisUser(username);
    }


    match /users/{username}/likes/{itemId} {
      allow read, create, update, delete: if request.auth != null
        && (isAdmin() || isThisUser(username) || request.resource.data.uid == request.auth.uid || resource.data.uid == request.auth.uid);
    }


    // AZOBSS 875: Sound Effects Favorites - one low-quota state document per user.
    match /users/{username}/soundFavorites/{docId} {
      allow read, create, update, delete: if request.auth != null
        && (isAdmin() || isThisUser(username) || request.resource.data.uid == request.auth.uid || resource.data.uid == request.auth.uid);
    }

    match /usernameAuthEmails/{username} {
      allow get: if true;
      allow list: if isAdmin();

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow update: if isAdmin()
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
          && request.resource.data.uid == request.auth.uid
        );

      allow delete: if isAdmin() || (
        request.auth != null
        && resource.data.uid == request.auth.uid
      );
    }

    match /settings/signupGate {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /settings/softwareStats {
      // One-document software stats bundle: downloads + ratings persist across deploys.
      // Public read/write is intentionally allowed for lightweight public counters.
      // Rating spam is limited in frontend with ratedBy browser token.
      allow read: if true;
      allow create, update: if true;
      allow delete: if isAdmin();
    }

    match /purchaseSummaries/{username} {
      allow read: if isAdmin() || isThisUser(username);

      allow create: if isAdmin() || (
        request.auth != null
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.usernameKey == username
      );

      allow update: if isAdmin() || (
        isThisUser(username)
        && resource.data.uid == request.auth.uid
        && resource.data.usernameKey == username
      );

      allow delete: if isAdmin();
    }


    match /purchaseRecords/{purchaseId} {
      allow read: if isAdmin()
        || (request.auth != null && resource.data.uid == request.auth.uid);

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow update, delete: if isAdmin();
    }

    match /purchaseLogs/{purchaseId} {
      allow read: if isAdmin()
        || (request.auth != null && resource.data.uid == request.auth.uid);

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow update, delete: if isAdmin();
    }

    match /siteStats/{docId} {
      allow read: if true;
      allow create, update: if true;
      allow delete: if isAdmin();
    }

    match /sitePresence/{visitorId} {
      allow read: if isAdmin();
      allow create, update: if true;
      allow delete: if isAdmin();
    }

    match /userLikes/{likeId} {
      allow read: if request.auth != null;

      allow create, update: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow delete: if request.auth != null
        && resource.data.uid == request.auth.uid;
    }

    match /onlineUsers/{uid} {
      allow read: if isAdmin();

      allow create, update: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow delete: if isAdmin()
        || (request.auth != null
        && resource.data.uid == request.auth.uid);
    }

    match /loginHistory/{docId} {
      allow read: if isAdmin();

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow update, delete: if isAdmin();
    }

    match /guestHistory/{docId} {
      allow read: if isAdmin();
      allow create: if true;
      allow update, delete: if isAdmin();
    }

    match /softwareTools/{itemId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /cadTools/{itemId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /affiliateProducts/{itemId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }



    // AZOBSS v1238: Support Inbox. Guests/users may submit; users can read only their own; admin manages all.
    match /supportMessages/{messageId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && (
            resource.data.uid == request.auth.uid
            || resource.data.userUid == request.auth.uid
            || resource.data.email == request.auth.token.email
            || resource.data.userEmail == request.auth.token.email
          )
        );
      allow create: if true;
      allow update, delete: if isAdmin();
    }

    // AZOBSS v1238: Site notifications are public-readable; only admin can publish/change/delete.
    match /notifications/{notificationId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    // AZOBSS v1238: Commission records. Admin can read/manage all; staff can read only records linked to their auth identity.
    match /commissionRecords/{recordId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && (
            resource.data.uid == request.auth.uid
            || resource.data.ownerUid == request.auth.uid
            || resource.data.staffUid == request.auth.uid
            || resource.data.sharerUid == request.auth.uid
            || resource.data.email == request.auth.token.email
            || resource.data.ownerEmail == request.auth.token.email
            || resource.data.staffEmail == request.auth.token.email
            || resource.data.sharerEmail == request.auth.token.email
          )
        );
      allow create, update, delete: if isAdmin();
    }

    // AZOBSS v1238: staff software submissions.
    match /staffSoftwareSubmissions/{submissionId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && (
            resource.data.uid == request.auth.uid
            || resource.data.ownerUid == request.auth.uid
            || resource.data.createdByUid == request.auth.uid
            || resource.data.staffUid == request.auth.uid
            || resource.data.email == request.auth.token.email
            || resource.data.ownerEmail == request.auth.token.email
            || resource.data.createdByEmail == request.auth.token.email
            || resource.data.staffEmail == request.auth.token.email
          )
        );
      allow create: if isAdmin()
        || (
          request.auth != null
          && request.resource.data.ownerUid == request.auth.uid
          && request.resource.data.createdByUid == request.auth.uid
          && isStaffUser(request.resource.data.ownerUsername)
        );
      allow update, delete: if isAdmin();
    }

    // AZOBSS v1238: staff CAD submissions.
    match /staffCADSubmissions/{submissionId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && (
            resource.data.uid == request.auth.uid
            || resource.data.ownerUid == request.auth.uid
            || resource.data.createdByUid == request.auth.uid
            || resource.data.staffUid == request.auth.uid
            || resource.data.email == request.auth.token.email
            || resource.data.ownerEmail == request.auth.token.email
            || resource.data.createdByEmail == request.auth.token.email
            || resource.data.staffEmail == request.auth.token.email
          )
        );
      allow create: if isAdmin()
        || (
          request.auth != null
          && request.resource.data.ownerUid == request.auth.uid
          && request.resource.data.createdByUid == request.auth.uid
          && isStaffUser(request.resource.data.ownerUsername)
        );
      allow update, delete: if isAdmin();
    }

    // AZOBSS v1238: explicit Food Orders access mapping managed from Admin > Users.
    match /foodOrderStaffAccess/{uid} {
      allow get: if isAdmin() || (request.auth != null && request.auth.uid == uid);
      allow list: if isAdmin();
      allow create, update, delete: if isAdmin();
    }

    // AZOBSS v1224: Marketplace listings (new + pre-owned)
    match /marketListings/{listingId} {
      allow read: if true;

      allow create: if request.auth != null
        && request.resource.data.sellerUid == request.auth.uid
        && request.resource.data.status == 'active';

      allow update: if isAdmin()
        || (request.auth != null
        && resource.data.sellerUid == request.auth.uid
        && request.resource.data.sellerUid == resource.data.sellerUid);

      allow delete: if isAdmin()
        || (request.auth != null
        && resource.data.sellerUid == request.auth.uid);
    }


    // AZOBSS v1244: Sales Partner applications.
    // A user who was previously approved as Staff may later be changed back to User.
    // That ordinary User must be able to submit a fresh pending application over the old approved record.
    function isCurrentSalesApplicantUser(username) {
      return request.auth != null
        && username is string
        && exists(/databases/$(database)/documents/users/$(username))
        && get(/databases/$(database)/documents/users/$(username)).data.uid == request.auth.uid
        && get(/databases/$(database)/documents/users/$(username)).data.role in [
          "user", "User", "customer", "Customer"
        ];
    }

    match /salesStaffApplications/{uid} {
      allow read: if isAdmin() || (request.auth != null && request.auth.uid == uid);

      allow create: if request.auth != null
        && request.auth.uid == uid
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == 'pending'
        && isCurrentSalesApplicantUser(request.resource.data.usernameKey);

      // v1244: current ordinary User may re-apply even when the existing application is historical status=approved.
      // The request still has to belong to the same Firebase UID and must reset the application to pending.
      allow update: if isAdmin() || (
        request.auth != null
        && request.auth.uid == uid
        && resource.data.uid == request.auth.uid
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == 'pending'
        && isCurrentSalesApplicantUser(request.resource.data.usernameKey)
      );

      allow delete: if isAdmin();
    }

    match /orders/{orderId} {
      allow read: if isAdmin()
        || (request.auth != null
        && resource.data.uid == request.auth.uid);

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow update, delete: if isAdmin();
    }

    match /downloadTokens/{tokenId} {
      allow read: if isAdmin();
      allow create, update, delete: if isAdmin();
    }

    match /receipts/{receiptId} {
      allow read: if isAdmin()
        || (request.auth != null
        && resource.data.uid == request.auth.uid);

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow update, delete: if isAdmin();
    }
  }
}
