rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {

    function isAdmin() {
      return request.auth != null
        && (
          request.auth.token.email == "zedan91@azobss.local"
          || request.auth.token.email == "zedan9107@gmail.com"
        );
    }

    function hasFoodOrderAccess() {
      return request.auth != null
        && firestore.exists(/databases/(default)/documents/foodOrderStaffAccess/$(request.auth.uid))
        && firestore.get(/databases/(default)/documents/foodOrderStaffAccess/$(request.auth.uid)).data.active == true
        && firestore.get(/databases/(default)/documents/foodOrderStaffAccess/$(request.auth.uid)).data.canViewFoodOrders == true;
    }

    function canManageFoodMenu() {
      return isAdmin() || hasFoodOrderAccess();
    }

    match /food-menu-images/{allPaths=**} {
      allow read: if true;

      allow create, update: if canManageFoodMenu()
        && request.resource.size < 5 * 1024 * 1024
        && request.resource.contentType.matches('image/(jpeg|png|webp)');

      allow delete: if canManageFoodMenu();
    }
  }
}
