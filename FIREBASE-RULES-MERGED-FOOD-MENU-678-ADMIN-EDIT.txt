rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return request.auth != null
        && (
          request.auth.token.email == "zedan91@azobss.local"
          || request.auth.token.email == "zedan9107@gmail.com"
        );
    }

    function isThisUser(username) {
      return request.auth != null
        && exists(/databases/$(database)/documents/users/$(username))
        && request.auth.uid == get(/databases/$(database)/documents/users/$(username)).data.uid;
    }

    function isStaffUser(username) {
      return request.auth != null
        && username is string
        && exists(/databases/$(database)/documents/users/$(username))
        && request.auth.uid == get(/databases/$(database)/documents/users/$(username)).data.uid
        && (
          get(/databases/$(database)/documents/users/$(username)).data.role in ["staff", "semiAdmin", "semiadmin", "semi-admin", "semi_admin", "seller", "editor"]
          || get(/databases/$(database)/documents/users/$(username)).data.permissions.canAccessStaffDashboard == true
          || get(/databases/$(database)/documents/users/$(username)).data.permissions.canManageSoftware == true
          || get(/databases/$(database)/documents/users/$(username)).data.permissions.canManageCAD == true
          || get(/databases/$(database)/documents/users/$(username)).data.permissions.canViewPayments == true
        );
    }

    // AZOBSS 671: explicit food-order access only.
    // Admin always has access. Every other account must have an active
    // foodOrderStaffAccess/{Firebase Auth UID} document created by admin.
    // Staff/semi-admin role alone does NOT grant access.
    function hasFoodOrderAccessDocument() {
      return request.auth != null
        && exists(/databases/$(database)/documents/foodOrderStaffAccess/$(request.auth.uid))
        && get(/databases/$(database)/documents/foodOrderStaffAccess/$(request.auth.uid)).data.keys().hasAll([
          "active", "canViewFoodOrders"
        ])
        && get(/databases/$(database)/documents/foodOrderStaffAccess/$(request.auth.uid)).data.active == true
        && get(/databases/$(database)/documents/foodOrderStaffAccess/$(request.auth.uid)).data.canViewFoodOrders == true;
    }

    function isFoodOrderStaff() {
      return isAdmin()
        || hasFoodOrderAccessDocument();
    }

    match /users/{username} {
      allow get: if true;
      allow list: if false;

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow update: if isAdmin()
        || isThisUser(username)
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
          && request.resource.data.uid == request.auth.uid
        );

      allow read, delete: if isAdmin() || isThisUser(username);
    }

    match /users/{username}/likes/{itemId} {
      allow read, create, update, delete: if request.auth != null
        && (
          isAdmin()
          || isThisUser(username)
          || request.resource.data.uid == request.auth.uid
          || resource.data.uid == request.auth.uid
        );
    }

    match /usernameAuthEmails/{username} {
      allow get: if true;
      allow list: if isAdmin();

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow update: if isAdmin()
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
          && request.resource.data.uid == request.auth.uid
        );

      allow delete: if isAdmin()
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
        );
    }

    match /settings/signupGate {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /settings/softwareStats {
      allow read: if true;
      allow create, update: if true;
      allow delete: if isAdmin();
    }

    match /purchaseSummaries/{username} {
      allow read: if isAdmin() || isThisUser(username);

      allow create: if isAdmin()
        || (
          request.auth != null
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.usernameKey == username
        );

      allow update: if isAdmin()
        || (
          isThisUser(username)
          && resource.data.uid == request.auth.uid
          && resource.data.usernameKey == username
        );

      allow delete: if isAdmin();
    }

    match /purchaseRecords/{purchaseId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
        );

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;

      allow update, delete: if isAdmin();
    }

    match /purchaseLogs/{purchaseId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
        );

      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status == "pending";

      allow update: if isAdmin()
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
          && request.resource.data.uid == resource.data.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
            "status",
            "paidAtMs",
            "paidAtClient",
            "downloadCount",
            "maxDownloads",
            "downloadExpiresAtMs",
            "downloadExpiresAtClient",
            "lastDownloadedAtMs",
            "lastDownloadedAtClient",
            "updatedAt"
          ])
          && request.resource.data.status in ["paid", "success", "completed", "settled"]
          && resource.data.status in ["pending", "paid", "success", "completed", "settled"]
          && request.resource.data.downloadCount is number
          && request.resource.data.downloadCount >= (resource.data.downloadCount is number ? resource.data.downloadCount : 0)
          && request.resource.data.downloadCount <= 5
        );

      allow delete: if isAdmin()
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
          && resource.data.status == "pending"
        );
    }

    match /siteStats/{docId} {
      allow read: if true;
      allow create, update: if true;
      allow delete: if isAdmin();
    }

    match /sitePresence/{visitorId} {
      allow read: if isAdmin();
      allow create, update: if true;
      allow delete: if isAdmin();
    }

    match /userLikes/{likeId} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null
        && request.resource.data.uid == request.auth.uid;
      allow delete: if request.auth != null
        && resource.data.uid == request.auth.uid;
    }

    match /onlineUsers/{uid} {
      allow read: if isAdmin();
      allow create, update: if request.auth != null
        && request.resource.data.uid == request.auth.uid;
      allow delete: if isAdmin()
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
        );
    }

    match /loginHistory/{docId} {
      allow read: if isAdmin();
      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    match /guestHistory/{docId} {
      allow read: if isAdmin();
      allow create: if true;
      allow update, delete: if isAdmin();
    }

    match /softwareTools/{itemId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /cadTools/{itemId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /affiliateProducts/{itemId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /orders/{orderId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
        );
      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    match /downloadTokens/{tokenId} {
      allow read: if isAdmin();
      allow create, update, delete: if isAdmin();
    }

    match /receipts/{receiptId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && resource.data.uid == request.auth.uid
        );
      allow create: if request.auth != null
        && request.resource.data.uid == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    match /supportMessages/{messageId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && (
            resource.data.uid == request.auth.uid
            || resource.data.userUid == request.auth.uid
            || resource.data.email == request.auth.token.email
            || resource.data.userEmail == request.auth.token.email
          )
        );
      allow create: if true;
      allow update, delete: if isAdmin();
    }

    match /notifications/{notificationId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /commissionRecords/{recordId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && (
            resource.data.uid == request.auth.uid
            || resource.data.ownerUid == request.auth.uid
            || resource.data.staffUid == request.auth.uid
            || resource.data.sharerUid == request.auth.uid
            || resource.data.email == request.auth.token.email
            || resource.data.ownerEmail == request.auth.token.email
            || resource.data.staffEmail == request.auth.token.email
            || resource.data.sharerEmail == request.auth.token.email
          )
        );
      allow create, update, delete: if isAdmin();
    }

    match /staffSoftwareSubmissions/{submissionId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && (
            resource.data.uid == request.auth.uid
            || resource.data.ownerUid == request.auth.uid
            || resource.data.createdByUid == request.auth.uid
            || resource.data.staffUid == request.auth.uid
            || resource.data.email == request.auth.token.email
            || resource.data.ownerEmail == request.auth.token.email
            || resource.data.createdByEmail == request.auth.token.email
            || resource.data.staffEmail == request.auth.token.email
          )
        );

      allow create: if isAdmin()
        || (
          request.auth != null
          && request.resource.data.ownerUid == request.auth.uid
          && request.resource.data.createdByUid == request.auth.uid
          && isStaffUser(request.resource.data.ownerUsername)
        );

      allow update, delete: if isAdmin();
    }

    match /staffCADSubmissions/{submissionId} {
      allow read: if isAdmin()
        || (
          request.auth != null
          && (
            resource.data.uid == request.auth.uid
            || resource.data.ownerUid == request.auth.uid
            || resource.data.createdByUid == request.auth.uid
            || resource.data.staffUid == request.auth.uid
            || resource.data.email == request.auth.token.email
            || resource.data.ownerEmail == request.auth.token.email
            || resource.data.createdByEmail == request.auth.token.email
            || resource.data.staffEmail == request.auth.token.email
          )
        );

      allow create: if isAdmin()
        || (
          request.auth != null
          && request.resource.data.ownerUid == request.auth.uid
          && request.resource.data.createdByUid == request.auth.uid
          && isStaffUser(request.resource.data.ownerUsername)
        );

      allow update, delete: if isAdmin();
    }

    // AZOBSS 671: explicit access mapping managed by Admin Dashboard.
    // Document ID must be the permitted user's Firebase Authentication UID.
    match /foodOrderStaffAccess/{uid} {
      allow get: if isAdmin()
        || (
          request.auth != null
          && request.auth.uid == uid
        );
      allow list: if isAdmin();
      allow create, update, delete: if isAdmin();
    }

    // AZOBSS 678: public brownie menu configuration.
    // Only Admin or users with explicit active Tempahan Makanan access
    // may edit names, prices, visibility and gallery metadata.
    match /foodMenuConfig/{configId} {
      allow read: if true;
      allow create, update, delete: if isFoodOrderStaff();
    }

    // AZOBSS 649: reliable food order records with customer phone.
    match /foodOrders/{orderId} {
      // Customers may place an order without logging in.
      // Creation is restricted to the exact fields and safe limits below.
      allow create: if
        request.resource.data.keys().hasOnly([
          "clientOrderId",
          "customerName",
          "customerPhone",
          "customerPhoneDigits",
          "requiredDate",
          "requiredDateLabel",
          "notes",
          "items",
          "totalBoxes",
          "totalPrice",
          "whatsappMessage",
          "whatsappNumber",
          "status",
          "source",
          "createdAt",
          "createdAtMs",
          "createdByUid",
          "createdByEmail"
        ])
        && request.resource.data.clientOrderId == orderId
        && request.resource.data.clientOrderId is string
        && request.resource.data.clientOrderId.size() >= 10
        && request.resource.data.clientOrderId.size() <= 100
        && request.resource.data.customerName is string
        && request.resource.data.customerName.size() >= 1
        && request.resource.data.customerName.size() <= 100
        && request.resource.data.customerPhone is string
        && request.resource.data.customerPhone.size() >= 8
        && request.resource.data.customerPhone.size() <= 20
        && request.resource.data.customerPhoneDigits is string
        && request.resource.data.customerPhoneDigits.matches('^[0-9]{8,15}$')
        && request.resource.data.requiredDate is string
        && request.resource.data.requiredDate.size() >= 8
        && request.resource.data.requiredDate.size() <= 20
        && request.resource.data.requiredDateLabel is string
        && request.resource.data.requiredDateLabel.size() <= 120
        && request.resource.data.notes is string
        && request.resource.data.notes.size() <= 800
        && request.resource.data.items is list
        && request.resource.data.items.size() >= 1
        && request.resource.data.items.size() <= 30
        && request.resource.data.totalBoxes is int
        && request.resource.data.totalBoxes >= 1
        && request.resource.data.totalBoxes <= 2970
        && request.resource.data.totalPrice is number
        && request.resource.data.totalPrice >= 1
        && request.resource.data.totalPrice <= 100000
        && request.resource.data.whatsappMessage is string
        && request.resource.data.whatsappMessage.size() <= 5000
        && request.resource.data.whatsappNumber == "60178809488"
        && request.resource.data.status == "new"
        && request.resource.data.source == "whatsapp"
        && request.resource.data.createdAt == request.time
        && request.resource.data.createdAtMs is int
        && request.resource.data.createdByUid is string
        && request.resource.data.createdByEmail is string
        && (
          (
            request.auth == null
            && request.resource.data.createdByUid == ""
            && request.resource.data.createdByEmail == ""
          )
          || (
            request.auth != null
            && request.resource.data.createdByUid == request.auth.uid
          )
        );

      // Only admin and accounts explicitly allowed by admin may view the records.
      allow read: if isFoodOrderStaff();

      // Explicitly allowed accounts may only change status and its audit fields.
      allow update: if isFoodOrderStaff()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          "status",
          "statusUpdatedAt",
          "statusUpdatedAtMs",
          "statusUpdatedByUid",
          "statusUpdatedByEmail"
        ])
        && request.resource.data.status in [
          "new", "contacted", "confirmed", "completed", "cancelled"
        ]
        && request.resource.data.statusUpdatedAt == request.time
        && request.resource.data.statusUpdatedAtMs is int
        && request.resource.data.statusUpdatedByUid == request.auth.uid
        && request.resource.data.statusUpdatedByEmail is string;

      // Keep records protected from accidental deletion. Admin only.
      allow delete: if isAdmin();
    }

    match /adminAuditLogs/{logId} {
      allow read, create, update, delete: if isAdmin();
    }
  }
}
