AZOBSS STAFF DIRECT MANAGEMENT FIRESTORE RULES ADDON

Tambahan/kemas kini diperlukan supaya semiAdmin/staff boleh edit terus collection sebenar:

match /softwareTools/{itemId} {
  allow read: if true;
  allow create, update, delete: if isAdmin()
    || (
      request.auth != null
      && exists(/databases/$(database)/documents/users/$(request.auth.uid))
      && (
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "semiAdmin"
        || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "staff"
      )
      && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.permissions.canManageSoftware == true
    );
}

match /cadTools/{itemId} {

Nota:
